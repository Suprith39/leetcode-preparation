118. Pascal's Triangle
class Solution {
    public List<List<Integer>> generate(int numRows) {

        List<List<Integer>> r= new ArrayList<>();
        for(int i=0;i<numRows;i++){
            List<Integer> row = new ArrayList<>();
            for(int j=0;j<i+1;j++){
                if (j == 0 || j == i){
                    row.add(1);

                }else{
                    row.add(r.get(i-1).get(j-1)+r.get(i-1).get(j));
                }
    
            }
            r.add(row);
        }
        return r;
    }
}
