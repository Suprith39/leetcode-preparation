    class Solution {
        public void moveZeroes(int[] nums) {
            int[] n=nums.clone();int k=0,c=nums.length;
            for(int i=0;i<nums.length;i++){
                if(n[i]==0){
                    c--;
                    continue;
                }
                else{
                    nums[k]=n[i];
                    k++;
                }
            }
            for(int i=c;i<nums.length;i++){
                nums[i]=0;
            }
        }
    }
