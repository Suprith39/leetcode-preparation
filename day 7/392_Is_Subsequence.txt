    class Solution {
        public boolean isSubsequence(String s, String t) {
            char[] c=t.toCharArray();
            char[] d=s.toCharArray();
            int k=0;
            for(int i=0;i<c.length;i++){
                if(k<d.length){
                    if(d[k]==c[i]){
                        k++;
                    }
                } else{
                    continue;
                }
            }
            if(k==d.length){
                return true;
            }else{
                return false;
            }
        }
    }
