class Solution { public boolean isPalindrome(String s) { s =
s.toUpperCase().replaceAll(“[^a-zA-Z0-9]”, ““).trim();

        char[] a = s.toCharArray();

        for (int i = 0; i < s.length() / 2; i++) {
            if (a[i] != a[s.length() - 1 - i]) {
                return false;
            }
        }

        return true;
    }

}
