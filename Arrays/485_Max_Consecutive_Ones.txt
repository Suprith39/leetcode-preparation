class Solution { public int findMaxConsecutiveOnes(int[] nums) { int ms
= 0; int s = 0;

        for (int i = 0; i < nums.length; i++) {
            if (nums[i] == 1) {
                s++;
            } else {
                s = 0;
            }

            if (s > ms) {
                ms = s;
            }
        }

        return ms;
    }

}
